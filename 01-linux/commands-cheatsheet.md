## Navigation & Inspection
pwd                   # current directory
ls -la <dir>          # list contents, incl. hidden, long format
cd <path>             # absolute path — from /
cd <relpath>          # relative path — from current dir
cd ..                 # up one level
cd ~                  # go to home directory
cat /proc/cpuinfo     # live CPU info from kernel
whereis <cmd>         # binary, source, man page locations
ps aux | grep <name>  # find running process + its launch args
